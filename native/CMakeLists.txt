cmake_minimum_required(VERSION 3.24)
project(EnhancedVehicles VERSION 0.0.66 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build with Visual Studio x64 on Windows.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_library(ev_minhook STATIC minhook/src/buffer.c minhook/src/hook.c minhook/src/trampoline.c minhook/src/hde/hde64.c)
target_include_directories(ev_minhook PUBLIC minhook/include)
add_library(EnhancedVehicles SHARED RV8NativeTrace.cpp)
target_compile_options(EnhancedVehicles PRIVATE /EHsc)
target_link_libraries(EnhancedVehicles PRIVATE ev_minhook shell32 ole32)
target_link_options(EnhancedVehicles PRIVATE /DYNAMICBASE /NXCOMPAT)
enable_testing()
add_executable(ev_native_fixtures filter-test.cpp)
target_compile_options(ev_native_fixtures PRIVATE /EHsc /UNDEBUG)
target_link_libraries(ev_native_fixtures PRIVATE ev_minhook shell32 ole32)
add_test(NAME native_fixtures COMMAND ev_native_fixtures)
